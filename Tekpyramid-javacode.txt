4th......public class Palindrome {
    public static void main(String[] args) {

        String name = "A man, a plan, a canal: Panama";

        int left = 0;
        int right = name.length() - 1;

        boolean isPalindrome = true;

        while (left < right) {

            // Skip spaces and punctuation from the left
            if (!Character.isLetterOrDigit(name.charAt(left))) {
                left++;
                continue;
            }

            // Skip spaces and punctuation from the right
            if (!Character.isLetterOrDigit(name.charAt(right))) {
                right--;
                continue;
            }

            // Compare characters ignoring case
            if (Character.toLowerCase(name.charAt(left)) !=
                Character.toLowerCase(name.charAt(right))) {

                isPalindrome = false;
                break;
            }

            left++;
            right--;
        }

        System.out.println(isPalindrome);
    }
}
